# Subfinder API Configuration Example
# Copy this file and set your API keys as environment variables

# === Free APIs (No registration required) ===
# These work without any API keys:
# - crt.sh
# - hackertarget  
# - wayback
# - anubis
# - alienvault

# === APIs requiring registration ===
# Set these environment variables for better results:

# BeVigil OSINT API
# Get from: https://bevigil.com/osint-api
# export BEVIGIL_API_KEY="your_api_key"

# BinaryEdge API
# Get from: https://www.binaryedge.io/
# export BINARYEDGE_API_KEY="your_api_key"

# BufferOver API
# Get from: https://tls.bufferover.run/
# export BUFFEROVER_API_KEY="your_api_key"

# BuiltWith API
# Get from: https://builtwith.com/api
# export BUILTWITH_API_KEY="your_api_key"

# C99 API
# Get from: https://api.c99.nl/
# export C99_API_KEY="your_api_key"

# Censys API
# Get from: https://censys.io/api
# export CENSYS_API_ID="your_api_id"
# export CENSYS_SECRET="your_secret"

# CertSpotter API
# Get from: https://sslmate.com/certspotter/api/
# export CERTSPOTTER_API_KEY="your_api_key"

# Chaos API
# Get from: https://chaos.projectdiscovery.io/
# export CHAOS_API_KEY="your_api_key"

# Chinaz API
# Get from: http://api.chinaz.com/
# export CHINAZ_API_KEY="your_api_key"

# DNSDB API
# Get from: https://www.farsightsecurity.com/dnsdb/

# DNSRepo API
# Get from: https://dnsrepo.noc.org/

# Facebook API
# Get from: https://developers.facebook.com/

# export FACEBOOK_APP_SECRET="your_app_secret"

# FOFA API
# Get from: https://fofa.so/api
# export FOFA_EMAIL="your_email"
# export FOFA_KEY="your_key"

# FullHunt API
# Get from: https://fullhunt.io/
# export FULLHUNT_API_KEY="your_api_key"

# GitHub Token
# Get from: https://github.com/settings/tokens

# Hunter API

# IntelX API

# === Windows Example ===
# set SHODAN_API_KEY=your_shodan_key
# set VIRUSTOTAL_API_KEY=your_vt_key
# python subfinder.py example.com

# === Linux/Mac Example ===
# export SHODAN_API_KEY="your_shodan_key"
# export VIRUSTOTAL_API_KEY="your_vt_key"
# python3 subfinder.py example.com
